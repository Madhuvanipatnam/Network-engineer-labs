!
!=========================================================
! Device : DSW2
! Role   : Distribution Layer Switch
!=========================================================
!

enable
configure terminal

hostname DSW2

no ip domain-lookup

ip routing

!
!=========================================================
! VLAN Configuration
!=========================================================
!

vlan 10
 name FINANCE

vlan 20
 name HR

vlan 30
 name SALES

vlan 40
 name IT

!
!=========================================================
! Trunk to Access Switch (SW2)
!=========================================================
!

interface Ethernet0/0
 description TO_SW2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 no shutdown

!
!=========================================================
! Layer 3 Links
!=========================================================
!

interface Ethernet0/3
 description TO_R2
 no switchport
 ip address 192.168.24.2 255.255.255.0
 no shutdown

interface Ethernet1/1
 description TO_R1
 no switchport
 ip address 192.168.14.2 255.255.255.0
 no shutdown

!
!=========================================================
! EtherChannel (LACP)
!=========================================================
!

interface range Ethernet0/1 - 2
 description TO_DSW1
 no switchport
 channel-group 1 mode active
 no shutdown

interface Port-channel1
 description L3_ETHERCHANNEL_TO_DSW1
 no switchport
 ip address 192.168.34.2 255.255.255.0
 no shutdown

!
!=========================================================
! VLAN Interfaces (SVIs)
!=========================================================
!

interface Vlan10
 description FINANCE_GATEWAY
 ip address 10.10.10.3 255.255.255.0
 standby 10 ip 10.10.10.1
 standby 10 priority 100
 standby 10 preempt
 no shutdown

interface Vlan20
 description HR_GATEWAY
 ip address 10.20.20.3 255.255.255.0
 standby 20 ip 10.20.20.1
 standby 20 priority 100
 standby 20 preempt
 no shutdown

interface Vlan30
 description SALES_GATEWAY
 ip address 10.30.30.3 255.255.255.0
 standby 30 ip 10.30.30.1
 standby 30 priority 100
 standby 30 preempt
 no shutdown

interface Vlan40
 description IT_GATEWAY
 ip address 10.40.40.3 255.255.255.0
 standby 40 ip 10.40.40.1
 standby 40 priority 100
 standby 40 preempt
 no shutdown

!
!=========================================================
! Spanning Tree
!=========================================================
!

spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40 root secondary

!
!=========================================================
! OSPF
!=========================================================
!

router ospf 1
 router-id 22.22.22.22

 network 10.10.10.0 0.0.0.255 area 0
 network 10.20.20.0 0.0.0.255 area 0
 network 10.30.30.0 0.0.0.255 area 0
 network 10.40.40.0 0.0.0.255 area 0

 network 192.168.14.0 0.0.0.255 area 0
 network 192.168.24.0 0.0.0.255 area 0
 network 192.168.34.0 0.0.0.255 area 0

end

write memory
