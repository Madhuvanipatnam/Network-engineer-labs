!
!=========================================================
! Device : R4
! Role   : ISP2 Router
! AS     : 400
!=========================================================
!

enable
configure terminal

hostname R4

no ip domain-lookup

!
!=========================================================
! Loopback Interface
!=========================================================
!

interface Loopback0
 description ISP2_ROUTER_ID
 ip address 4.4.4.4 255.255.255.255

!
!=========================================================
! Interfaces
!=========================================================
!

interface GigabitEthernet0/0
 description TO_R2
 ip address 198.51.100.2 255.255.255.0
 no shutdown

interface GigabitEthernet1/0
 description TO_INTERNET
 ip address 192.0.3.1 255.255.255.0
 no shutdown

!
!=========================================================
! BGP Configuration
!=========================================================
!

router bgp 400
 bgp log-neighbor-changes

 !
 ! Enterprise
 !
 neighbor 198.51.100.1 remote-as 100

 !
 ! Internet
 !
 neighbor 192.0.3.2 remote-as 65010

 !
 ! Networks to Advertise
 !
 network 4.4.4.4 mask 255.255.255.255
 network 198.51.100.0 mask 255.255.255.0

end

write memory
