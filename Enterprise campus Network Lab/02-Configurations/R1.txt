!
!=========================================================
! Device : R1
! Role   : Enterprise Edge Router
!=========================================================
!

enable
configure terminal

hostname R1

no ip domain-lookup

!
!=========================================================
! Loopback Interface
!=========================================================
!

interface Loopback0
 description ROUTER_ID
 ip address 1.1.1.1 255.255.255.255

!
!=========================================================
! Interfaces
!=========================================================
!

interface GigabitEthernet0/0
 description TO_R2
 ip address 192.168.12.1 255.255.255.0
 no shutdown

interface GigabitEthernet1/0
 description TO_ISP1
 ip address 203.0.113.1 255.255.255.0
 ip nat outside
 no shutdown

interface Ethernet5/0
 description TO_DSW1
 ip address 192.168.13.1 255.255.255.0
 ip nat inside
 no shutdown

interface Ethernet5/1
 description TO_DSW2
 ip address 192.168.14.1 255.255.255.0
 ip nat inside
 no shutdown

!
!=========================================================
! OSPF
!=========================================================
!

router ospf 1
 router-id 1.1.1.1

 network 1.1.1.1 0.0.0.0 area 0
 network 192.168.12.0 0.0.0.255 area 0
 network 192.168.13.0 0.0.0.255 area 0
 network 192.168.14.0 0.0.0.255 area 0

 default-information originate

!
!=========================================================
! Static Default Route
!=========================================================
!

ip route 0.0.0.0 0.0.0.0 203.0.113.2

!
!=========================================================
! NAT (PAT)
!=========================================================
!

access-list 1 permit 10.10.10.0 0.0.0.255
access-list 1 permit 10.20.20.0 0.0.0.255
access-list 1 permit 10.30.30.0 0.0.0.255
access-list 1 permit 10.40.40.0 0.0.0.255

ip nat inside source list 1 interface GigabitEthernet1/0 overload

!
!=========================================================
! BGP
!=========================================================
!

router bgp 100
 bgp log-neighbor-changes

 neighbor 203.0.113.2 remote-as 300

 network 1.1.1.1 mask 255.255.255.255
 network 203.0.113.0 mask 255.255.255.0

end

write memory
