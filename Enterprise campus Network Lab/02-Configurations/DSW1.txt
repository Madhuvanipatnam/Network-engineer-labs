!
!=========================================================
! Device : DSW1
! Role   : Distribution Layer Switch
!=========================================================
!

enable
configure terminal

hostname DSW1

no ip domain-lookup

ip routing

!
!=========================================================
! VLAN Configuration
!=========================================================
!

vlan 10
 name FINANCE

vlan 20
 name HR

vlan 30
 name SALES

vlan 40
 name IT

!
!=========================================================
! Trunk to SW1
!=========================================================
!

interface Ethernet0/0
 description TO_SW1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 no shutdown

!
!=========================================================
! Layer 3 Links
!=========================================================
!

interface Ethernet0/3
 description TO_R1
 no switchport
 ip address 192.168.13.2 255.255.255.0
 no shutdown

interface Ethernet1/1
 description TO_R2
 no switchport
 ip address 192.168.23.2 255.255.255.0
 no shutdown

!
!=========================================================
! LACP EtherChannel
!=========================================================
!

interface range Ethernet0/1 - 2
 description TO_DSW2
 no switchport
 channel-group 1 mode active
 no shutdown

interface Port-channel1
 description L3_ETHERCHANNEL_TO_DSW2
 no switchport
 ip address 192.168.34.1 255.255.255.0
 no shutdown

!
!=========================================================
! VLAN Interfaces
!=========================================================
!

interface Vlan10
 description FINANCE_GATEWAY
 ip address 10.10.10.2 255.255.255.0
 standby 10 ip 10.10.10.1
 standby 10 priority 110
 standby 10 preempt
 no shutdown

interface Vlan20
 description HR_GATEWAY
 ip address 10.20.20.2 255.255.255.0
 standby 20 ip 10.20.20.1
 standby 20 priority 110
 standby 20 preempt
 no shutdown

interface Vlan30
 description SALES_GATEWAY
 ip address 10.30.30.2 255.255.255.0
 standby 30 ip 10.30.30.1
 standby 30 priority 110
 standby 30 preempt
 no shutdown

interface Vlan40
 description IT_GATEWAY
 ip address 10.40.40.2 255.255.255.0
 standby 40 ip 10.40.40.1
 standby 40 priority 110
 standby 40 preempt
 no shutdown

!
!=========================================================
! DHCP Configuration
!=========================================================
!

ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.20.20.1 10.20.20.20
ip dhcp excluded-address 10.30.30.1 10.30.30.20
ip dhcp excluded-address 10.40.40.1 10.40.40.20

ip dhcp pool VLAN10
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 8.8.8.8

ip dhcp pool VLAN20
 network 10.20.20.0 255.255.255.0
 default-router 10.20.20.1
 dns-server 8.8.8.8

ip dhcp pool VLAN30
 network 10.30.30.0 255.255.255.0
 default-router 10.30.30.1
 dns-server 8.8.8.8

ip dhcp pool VLAN40
 network 10.40.40.0 255.255.255.0
 default-router 10.40.40.1
 dns-server 8.8.8.8

!
!=========================================================
! Rapid PVST
!=========================================================
!

spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40 root primary

!
!=========================================================
! OSPF Configuration
!=========================================================
!

router ospf 1
 router-id 11.11.11.11

 network 10.10.10.0 0.0.0.255 area 0
 network 10.20.20.0 0.0.0.255 area 0
 network 10.30.30.0 0.0.0.255 area 0
 network 10.40.40.0 0.0.0.255 area 0

 network 192.168.13.0 0.0.0.255 area 0
 network 192.168.23.0 0.0.0.255 area 0
 network 192.168.34.0 0.0.0.255 area 0

end

write memory
